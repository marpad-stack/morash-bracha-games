let ac=null;
function beep(f,d,type){
  if(!snd)return;
  try{
    ac=ac||new (window.AudioContext||window.webkitAudioContext)();
    const o=ac.createOscillator(),g=ac.createGain();
    o.type=type||'sine';o.frequency.value=f;o.connect(g);g.connect(ac.destination);
    g.gain.setValueAtTime(.14,ac.currentTime);
    g.gain.exponentialRampToValueAtTime(.001,ac.currentTime+d);
    o.start();o.stop(ac.currentTime+d);
  }catch(e){}
}

const cvs=document.getElementById('cft'),ctx=cvs.getContext('2d');
let parts=[],running=false;
function size(){cvs.width=innerWidth;cvs.height=innerHeight}

const COL=['#E8836F','#1F7A8C','#E5B25D','#F7E9C9','#D96A54','#FFFFFF'];
function burst(n,pw){
  for(let i=0;i<n;i++){
    parts.push({x:innerWidth/2+(Math.random()-.5)*150,y:innerHeight*.35,
      vx:(Math.random()-.5)*pw,vy:-(Math.random()*pw*.9+4),g:.18+Math.random()*.12,
      s:6+Math.random()*7,r:Math.random()*6,vr:(Math.random()-.5)*.3,
      c:COL[i%COL.length],life:120+Math.random()*40});
  }
  if(!running){running=true;loop()}
}
function loop(){
  ctx.clearRect(0,0,cvs.width,cvs.height);
  parts.forEach(function(p){
    p.vy+=p.g;p.x+=p.vx;p.y+=p.vy;p.r+=p.vr;p.life--;
    ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.r);ctx.fillStyle=p.c;
    ctx.fillRect(-p.s/2,-p.s/2,p.s,p.s*.6);ctx.restore();
  });
  parts=parts.filter(function(p){return p.life>0&&p.y<innerHeight+40});
  if(parts.length)requestAnimationFrame(loop);
  else{running=false;ctx.clearRect(0,0,cvs.width,cvs.height)}
}
