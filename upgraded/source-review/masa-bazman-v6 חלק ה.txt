function renderStep(){
  misses=0;busy=false;
  $('stepLabel').textContent=L(STR.step).replace('%n',step+1)
    + (players===2 ? ' · '+L(STR.turn).replace('%n',turn+1) : '');

function answer(idx,el){
  if(busy)return;
  if(idx!==step){
    misses++;streak=0;sBad();
    if(el){el.classList.add('wrong');setTimeout(()=>el.classList.remove('wrong'),470)}
    $('fb').className='fb bad';
    $('fb').textContent = misses>=hintAfter()?L(STR.hint):L(STR.wrong);
    if(misses>=hintAfter()){
      const r=$('hand').querySelector('.card[data-i="'+step+'"]');
      if(r)r.classList.add('glow');
    }
    return;
  }
  busy=true;
  const clean = misses===0;
  if(clean){scores[turn]++;streak++} else streak=0;
  renderTurn();sGood();
  $('fb').className='fb good';
  $('fb').textContent=clean?L(STR.great):L(STR.ok);
  $('hand').querySelectorAll('.card').forEach(c=>c.classList.add('gone'));
  sparkle(el);
  if(streak>=3)combo(streak);
  setTimeout(()=>runMini(step),620);
}

function afterMini(){
  addHouseItem(step);
  step++;
  if(players===2) turn=1-turn;
  if(step>=N) finish(); else renderStep();
}

/* --- החלקה קדימה ואחורה --- */
/* ההתקדמות נמדדת לפי המרחק שהאצבע עברה, לא לפי זיהוי היפוכי כיוון.
   במגע כל אירוע תנועה הוא זעיר, ולכן ספירת היפוכים כמעט אף פעם לא נרשמה. */
function miniScrub(m,area){
  area.appendChild(el('mtrack',''));
  const it=el('mitem cue',m.item,{left:'50%',top:'46%'});
  area.appendChild(it);
  area.appendChild(el('mhint','↔ '+L(STR.hScrub)));

  let down=false,lastX=null,dist=0,dir=0;
  let need=area.getBoundingClientRect().width*2.2||600;

  const bubble=x=>{
    const b=el('bub','',{left:x+'px',top:'46%'});
    area.appendChild(b);setTimeout(()=>b.remove(),780);
  };
  const move=e=>{
    const r=area.getBoundingClientRect();
    const x=Math.max(r.left+16,Math.min(r.right-16,e.clientX));
    it.style.left=((x-r.left)/r.width*100)+'%';
    if(lastX!==null){
      const dx=x-lastX;
      dist+=Math.abs(dx);
      const d=Math.sign(dx);
      if(d!==0&&d!==dir&&Math.abs(dx)>2){dir=d;sTick();bubble(x-r.left)}
      bar(dist/need*100);
      if(dist>=need)miniFinish();
    }
    lastX=x;
  };
  const start=e=>{
    e.preventDefault();down=true;lastX=null;
    it.classList.remove('cue');
    need=area.getBoundingClientRect().width*2.2;
    try{area.setPointerCapture(e.pointerId)}catch(_){}
    move(e);
  };
  const end=()=>{down=false;lastX=null};

  area.addEventListener('pointerdown',start);
  area.addEventListener('pointermove',e=>{if(down){e.preventDefault();move(e)}});
  area.addEventListener('pointerup',end);
  area.addEventListener('pointercancel',end);
  /* גיבוי למכשירים שלא שולחים אירועי pointer */
  area.addEventListener('touchmove',e=>{
    if(!e.touches[0])return;
    e.preventDefault();
    if(!down){down=true;lastX=null;it.classList.remove('cue')}
    move(e.touches[0]);
  },{passive:false});
  area.addEventListener('touchend',end);
}

/* --- גרירה אל יעד --- */
function miniDrag(m,area){
  const tg=el('mtarget',m.to,{left:'22%',top:'50%'});
  const it=el('mitem cue',m.item,{left:'78%',top:'50%'});
  area.appendChild(tg);area.appendChild(it);
  area.appendChild(el('mhint','← '+L(STR.hDrag)));
  let down=false;
  const near=()=>{
    const a=it.getBoundingClientRect(),b=tg.getBoundingClientRect();
    return Math.hypot(a.left+a.width/2-(b.left+b.width/2),
                      a.top+a.height/2-(b.top+b.height/2))<58;
  };
  const move=e=>{
    const p=pos(e,area);it.style.left=p.x+'%';it.style.top=p.y+'%';
    const n=near();tg.classList.toggle('hot',n);
    bar(n?88:Math.min(70,(78-p.x)/56*80));
    return n;
  };
  area.addEventListener('pointerdown',e=>{
    e.preventDefault();down=true;it.classList.remove('cue');
    try{area.setPointerCapture(e.pointerId)}catch(_){}
    move(e);
  });
  area.addEventListener('pointermove',e=>{if(down){e.preventDefault();move(e)}});
  const up=()=>{
    if(!down)return;down=false;
    if(near())miniFinish();
    else{it.style.left='78%';it.style.top='50%';it.classList.add('cue');bar(0)}
  };
  area.addEventListener('pointerup',up);
  area.addEventListener('pointercancel',up);
}

function miniHold(m,area){
  area.insertAdjacentHTML('beforeend',
    '<svg class="mring" viewBox="0 0 100 100"><circle class="bg" cx="50" cy="50" r="45"/>'+
    '<circle class="fg" cx="50" cy="50" r="45"/></svg>');
  const fg=area.querySelector('.fg');
  const it=el('mitem cue',m.item,{left:'50%',top:'50%',fontSize:'2.6rem'});
  area.appendChild(it);
  area.appendChild(el('mhint',L(STR.hHold)));
  let raf=null,t0=0;
  const loop=()=>{
    const p=Math.min(1,(performance.now()-t0)/m.n);
    bar(p*100);fg.style.strokeDashoffset=283*(1-p);
    it.style.transform='translate(-50%,-50%) scale('+(1+p*.55)+')';
    if(p>=1)miniFinish(); else raf=requestAnimationFrame(loop);
  };
  const stop=()=>{
    cancelAnimationFrame(raf);
    if(miniDone)return;
    bar(0);fg.style.strokeDashoffset=283;it.style.transform='translate(-50%,-50%)';
  };
  area.addEventListener('pointerdown',e=>{
    e.preventDefault();it.classList.remove('cue');
    t0=performance.now();cancelAnimationFrame(raf);raf=requestAnimationFrame(loop);
  });
  area.addEventListener('pointerup',stop);
  area.addEventListener('pointercancel',stop);
  area.addEventListener('pointerleave',stop);
}
