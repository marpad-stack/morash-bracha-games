/* ---------------- generic drag helper ---------------- */
function makeDraggable(el, targets, {consume=false, onDrop=()=>{}} = {}){
  let ox=0, oy=0, sx=0, sy=0, curx=0, cury=0, active=false, locked=false;
  el.style.touchAction='none';
  el.addEventListener('pointerdown', e=>{
    if(locked) return;
    el.classList.remove('pulse-hint');
    active=true; try{ el.setPointerCapture(e.pointerId); }catch(_){}
    sx=e.clientX; sy=e.clientY; el.style.transition='none'; el.classList.add('dragging');
    targets.forEach(t=>t.classList.add('active'));
  });
  el.addEventListener('pointermove', e=>{
    if(!active) return;
    curx = ox + (e.clientX-sx); cury = oy + (e.clientY-sy);
    el.style.transform = `translate(${curx}px,${cury}px)`;
  });
  function release(){
    if(!active) return;
    active=false; el.classList.remove('dragging');
    targets.forEach(t=>t.classList.remove('active'));
    el.style.transition='transform .32s cubic-bezier(.2,.8,.3,1.15)';
    const er = el.getBoundingClientRect();
    const ecx = er.left+er.width/2, ecy = er.top+er.height/2;
    let hit=null;
    for(const t of targets){
      const tr=t.getBoundingClientRect();
      if(ecx>tr.left && ecx<tr.right && ecy>tr.top && ecy<tr.bottom){ hit=t; break; }
    }
    if(hit){
      const tr=hit.getBoundingClientRect();
      const dx=(tr.left+tr.width/2)-ecx, dy=(tr.top+tr.height/2)-ecy;
      curx+=dx; cury+=dy; ox=curx; oy=cury;
      el.style.transform = `translate(${curx}px,${cury}px)`;
      locked=true;
      if(consume){ setTimeout(()=>{ el.style.opacity='0'; el.style.transform+=' scale(.3)'; },60); }
      onDrop(true,hit);
    } else {
      curx=ox; cury=oy; el.style.transform=`translate(${ox}px,${oy}px)`;
      onDrop(false,null);
    }
  }
  el.addEventListener('pointerup', release);
  el.addEventListener('pointercancel', release);
}

/* ---------------- engine ---------------- */
const TOTAL_TIME = 240;
let idx=0, left=TOTAL_TIME, tick=null, timeUp=false, lastWon=null, gameActive=false;

function fmt(sec){ const m=Math.floor(sec/60), s=sec%60; return m+':'+String(s).padStart(2,'0'); }
function paintClock(){
  $('clock').textContent = fmt(Math.max(0,left));
  const pct = Math.max(0, left/TOTAL_TIME)*100;
  $('wax').style.height = pct+'%';
  $('flame').style.bottom = 'calc('+(pct*0.88)+'px + 6px)';
  $('candle').classList.toggle('hurry', left<=30);
}
function show(id){ document.querySelectorAll('.screen').forEach(s=>s.classList.remove('on')); $(id).classList.add('on'); }
function buildDots(){ $('dots').innerHTML = SCENES.map((_,k)=>`<span class="dot" data-k="${k}"></span>`).join(''); }
function say(text, good){ const b=$('banner'); b.textContent=text; b.classList.toggle('good', !!good); }

function renderDoneList(){
  const list = $('done');
  list.innerHTML='';
  for(let k=0;k<idx;k++){
    const sc=SCENES[k];
    list.insertAdjacentHTML('beforeend',
      `<li class="done-item"><span class="num">${k+1}</span><span class="txt">${L(sc.t)}<span class="say">${L(sc.say)}</span></span></li>`);
  }
  $('empty').style.display = idx===0 ? '' : 'none';
}
function renderStaticText(){
  $('txt-eyebrow').textContent = L(G.eyebrow);
  $('txt-title').textContent = L(G.title);
  $('txt-sub').textContent = L(G.sub);
  $('txt-p1').textContent = L(G.p1);
  $('txt-tip').textContent = L(G.tip);
  $('btn-start').textContent = L(G.startBtn);
  $('btn-again').textContent = L(G.againBtn);
  $('txt-h2').textContent = L(G.sub);
  $('txt-doneHeading').textContent = L(G.doneHeading);
  $('empty').textContent = L(G.emptyMsg);
  const nb = NIKUD ? L(G.nikudOff) : L(G.nikudOn);
  $('nikudBtn1').textContent = '🔤 ' + nb;
  $('nikudBtn2').textContent = '🔤 ' + nb;
  $('nikudBtn1').classList.toggle('on', NIKUD);
  $('nikudBtn2').classList.toggle('on', NIKUD);
  $('count-meta').textContent = idx + ' ' + L(G.countSuffix);
  renderDoneList();
}

function loadScene(){
  const sc = SCENES[idx];
  $('stage-title').textContent = L(sc.t);
  say(L(sc.instr), false);
  const stage = $('stage'); stage.innerHTML='';
  sc.build(stage, onSceneDone, sc);
}
function onSceneDone(){
  const sc = SCENES[idx];
  say(L(sc.say), true);
  idx++;
  $('count-meta').textContent = idx + ' ' + L(G.countSuffix);
  renderDoneList();
  const dotEls = document.querySelectorAll('.dot');
  if(dotEls[idx-1]) dotEls[idx-1].classList.add('done');
  if(idx>=SCENES.length){ finish(true); return; }
  setTimeout(loadScene, 900);
}
function finish(won){
  clearInterval(tick);
  gameActive=false;
  lastWon = {won, timeUp, idx, left};
  renderEnding();
  show('screen-end');
}
function renderEnding(){
  if(!lastWon) return;
  const {won, timeUp, idx, left} = lastWon;
  if(won && !timeUp){
    $('end-title').textContent = L(G.wonTitle);
    $('end-score').textContent = L(G.wonScorePre) + fmt(left) + L(G.wonScorePost);
    $('end-text').textContent = L(G.wonText);
  } else if(won && timeUp){
    $('end-title').textContent = L(G.lateTitle);
    $('end-score').textContent = L(G.lateScoreDone);
    $('end-text').textContent = L(G.lateText);
  } else {
    $('end-title').textContent = L(G.lateTitle);
    $('end-score').textContent = idx + ' ' + L(G.countSuffix);
    $('end-text').textContent = L(G.lateText);
  }
}
function start(){
  idx=0; left=TOTAL_TIME; timeUp=false; gameActive=true; lastWon=null;
  buildDots(); paintClock();
  renderStaticText();
  loadScene();
  show('screen-game');
  clearInterval(tick);
  tick=setInterval(()=>{
    left--; paintClock();
    if(left<=0){ timeUp=true; left=0; paintClock(); clearInterval(tick); }
  },1000);
}
function toggleNikud(){
  NIKUD = !NIKUD;
  renderStaticText();
  if(gameActive){ loadScene(); }
  if(lastWon){ renderEnding(); }
}
$('btn-start').addEventListener('click', start);
$('btn-again').addEventListener('click', start);
$('nikudBtn1').addEventListener('click', toggleNikud);
$('nikudBtn2').addEventListener('click', toggleNikud);
