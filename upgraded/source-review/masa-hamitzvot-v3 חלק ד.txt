function hop(steps){
  const p=cur();let n=0;
  const step=()=>{
    p.pos=(p.pos+1)%BOARD.length;n++;
    placeTokens();
    const c=cellEls[p.pos];c.classList.remove("hot");void c.offsetWidth;c.classList.add("hot");
    if(n<steps)setTimeout(step,230);
    else setTimeout(()=>land(BOARD[p.pos]),320);
  };
  setTimeout(step,180);
}
function land(type){
  const p=cur();
  if(p.ai)return aiResolve(type,p);
  if(type==="mitzva")cardMitzva(p);
  else if(type==="sit")cardSit(p);
  else if(type==="quiz")cardQuiz(p);
  else if(type==="challenge")cardChal(p);
  else if(type==="chesed")cardChesed(p);
  else if(type==="shabbat")cardShabbat(p);
  else if(type==="surprise")cardSurprise(p);
  else nextTurn();
}

function aiResolve(type,p){
  const c=cityOf(p);
  let msg,pts=0,mm=null;
  if(type==="mitzva"){
    const pool=MITZVOT.filter(m=>!c.done[m.id]);
    mm=pool.length?rnd(pool):rnd(MITZVOT);
    c.done[mm.id]=(c.done[mm.id]||0)+1;pts=mm.p;
    msg=K("השלים את המשימה ובנה ","הִשְׁלִים אֶת הַמְּשִׂימָה וּבָנָה ")+K(mm.b,mm.bn);
  }else if(type==="sit"){pts=2;msg=K("בחר בדרך הטובה","בָּחַר בַּדֶּרֶךְ הַטּוֹבָה");}
  else if(type==="quiz"){const ok=Math.random()<.7;pts=ok?2:0;msg=ok?K("ענה נכון!","עָנָה נָכוֹן!"):K("טעה בשאלה","טָעָה בַּשְּׁאֵלָה");}
  else if(type==="challenge"){pts=2;msg=K("עמד באתגר","עָמַד בָּאֶתְגָּר");}
  else if(type==="chesed"){const o=rnd(G.players.filter(x=>x!==p));if(o)o.pts+=2;pts=2;msg=K("עשה חסד ונתן נקודות","עָשָׂה חֶסֶד וְנָתַן נְקֻדּוֹת");}
  else if(type==="shabbat"){G.players.forEach(x=>x.pts+=1);msg=K("הדליק נרות — שבת שלום לכולם","הִדְלִיק נֵרוֹת — שַׁבָּת שָׁלוֹם לְכֻלָּם");}
  else if(type==="surprise"){pts=3;msg=K("קיבל הפתעה","קִבֵּל הַפְתָּעָה");}
  else msg=K("עבר את ההתחלה","עָבַר אֶת הַהַתְחָלָה");
  p.pts+=pts;
  renderGame();if(mm)renderMini(mm.id);
  showSheet('<div class="kicker">'+p.name+'</div>'+
    (mm?'<div style="height:88px;display:flex;align-items:flex-end;justify-content:center;overflow:hidden"><div style="transform:scale(.48);transform-origin:bottom center">'+drawB(mm)+'</div></div>':'<div style="margin:8px 0">'+ic("crane",48)+'</div>')+
    '<div class="cardtxt">'+msg+'</div>'+(pts?'<div class="gain">+'+pts+'</div>':''));
  setTimeout(()=>{closeSheet();nextTurn();},1800);
}

const cv=$("#fx"),ctx=cv.getContext("2d");
let parts=[],running=false;
function fit(){cv.width=innerWidth;cv.height=innerHeight;}
addEventListener("resize",fit);fit();
function confetti(){
  const cols=["#e9408f","#f2b705","#38b6ff","#3fd0a8","#ff7a6b","#a78bfa"];
  for(let i=0;i<80;i++)parts.push({x:innerWidth/2+(Math.random()-.5)*150,y:innerHeight*.35,
    vx:(Math.random()-.5)*9,vy:-Math.random()*12-4,s:5+Math.random()*6,c:cols[i%cols.length],
    r:Math.random()*6,vr:(Math.random()-.5)*.4,l:0});
  loop();
}
function loop(){
  if(running)return;running=true;
  (function f(){
    ctx.clearRect(0,0,cv.width,cv.height);
    parts=parts.filter(p=>p.l<120);
    parts.forEach(p=>{
      p.l++;p.vy+=.42;p.x+=p.vx;p.y+=p.vy;p.r+=p.vr;
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(p.r);
      ctx.globalAlpha=Math.max(0,1-p.l/120);
      ctx.fillStyle=p.c;ctx.fillRect(-p.s/2,-p.s/2,p.s,p.s*.6);ctx.restore();
    });
    if(parts.length)requestAnimationFrame(f);
    else{ctx.clearRect(0,0,cv.width,cv.height);running=false;}
  })();
}
