function buildText(){
  var all=activeTasks(),done=all.filter(function(t){return isDone(idOf(t))}).length;
  var L=['*'+(P.name?'המפה של '+P.name:'המפה שלי')+' · שפרה ופועה*',done+' מתוך '+all.length+' צעדים כבר מאחוריי'];
  if(P.birth)L.push(baby()+' · '+fmtFull(new Date(P.birth)));L.push('');
  var open=all.filter(function(t){return !isDone(idOf(t))});if(!open.length){L.push('הכול סומן 🌟');return L.join('\n');}
  L.push('*מה שעוד לפניי:*');var g={};open.forEach(function(t){(g[t.cat]=g[t.cat]||[]).push(t);});
  Object.keys(g).forEach(function(k){L.push('','*'+k+'*');g[k].forEach(function(t){var s=statusOf(t);L.push((t.u?'⭐ ':'• ')+fill(t.t)+' ('+(s?s.txt:(t.w||''))+')'+(data.notes[idOf(t)]?'\n   ↳ '+data.notes[idOf(t)]:''));});});
  return L.join('\n');
}
